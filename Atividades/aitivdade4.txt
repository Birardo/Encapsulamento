namespace Encapsulamento
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Pessoa p = new Pessoa();
            p.Nome = "João";
            Console.WriteLine(p.Nome);

            //Atividade 2 


            Produto t = new Produto();

            Console.WriteLine(t.preco);

            c.frear();
            Console.WriteLine(c.velocidadeAtual);

            Console.WriteLine(c.velocidadeAtual);
            c.frear();

            //c.velocidadeAtual = 10;
            // erro: propriedade ou o indexador "Carro.velocidadeAtual" não pode ser atribuído, pois é somente leitura

            Console.WriteLine(temperatura.celcius);

            Console.WriteLine(temperatura.kelvin);
